      // Un clic (ou un tap) retourne le sachet, un second clic le remet de face
      let isBack = false, flipping = false, ft = 0;
      const showBack = (on) => {
        isBack = on;
        base = on ? rest + 180 : rest;
        pouch.classList.remove("is-live");
        setAngle(pouch, base);
        if (turn) { turn.setAttribute("aria-pressed", String(on)); label.textContent = on ? turn.dataset.back : turn.dataset.front; }
      };
      const flip = () => {
        flipping = true; clearTimeout(ft); ft = setTimeout(() => { flipping = false; }, 900);
        showBack(!isBack);
      };
      if (turn) turn.addEventListener("click", flip);
      pouch.addEventListener("click", flip);
      pouch.addEventListener("keydown", (e) => {
        if (e.key === "Enter" || e.key === " ") { e.preventDefault(); flip(); return; }
        if (e.key === "ArrowLeft" || e.key === "ArrowRight") { e.preventDefault(); base += e.key === "ArrowLeft" ? -20 : 20; setAngle(pouch, base); }
      });
      if (reduce) return;
      // Survol à la souris : légère inclinaison vers le curseur
      pouch.addEventListener("pointermove", (e) => {
        if (e.pointerType !== "mouse" || flipping) return;
        const r = pouch.getBoundingClientRect();
        const x = (e.clientX - r.left) / r.width - 0.5, y = (e.clientY - r.top) / r.height - 0.5;
        pouch.classList.add("is-live");
        setAngle(pouch, base + x * 16, y * -8);
      });
      pouch.addEventListener("pointerleave", () => { pouch.classList.remove("is-live"); setAngle(pouch, base); });
